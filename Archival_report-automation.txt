        ID INT IDENTITY(1,1) PRIMARY KEY,

        Source_Table_Name VARCHAR(500),

CREATE OR ALTER PROC dbo.usp_Generate_Monthly_Archival_Report
(
    @ActivityDate DATE
)
AS
BEGIN

SET NOCOUNT ON;

DECLARE
    @SourceDB SYSNAME,
    @SourceSchema SYSNAME,
    @SourceTable SYSNAME,
    @TargetDB SYSNAME,
    @TargetSchema SYSNAME,
    @TargetTable SYSNAME,
    @DateColumn SYSNAME,
    @RetentionType VARCHAR(10),
    @RetentionValue INT,

    @FromDate DATETIME,
    @ToDate DATETIME,
    @BeforeCount BIGINT,
    @ArchivedRows BIGINT,
    @AfterCount BIGINT,

    @SQL NVARCHAR(MAX)

DECLARE Cur CURSOR FAST_FORWARD FOR
SELECT
    Source_DB,
    Source_Schema,
    Source_Table,
    Target_DB,
    Target_Schema,
    Target_Table,
    Date_Column,
    Retention_Type,
    Retention_Value
FROM dbo.Archival_Config
WHERE IsActive=1

OPEN Cur

FETCH NEXT FROM Cur INTO
@SourceDB,@SourceSchema,@SourceTable,
@TargetDB,@TargetSchema,@TargetTable,
@DateColumn,@RetentionType,@RetentionValue

WHILE @@FETCH_STATUS=0
BEGIN

    ---------------------------------------------------
    -- From Date (Last Archived Date)
    ---------------------------------------------------

    SET @SQL='
    SELECT @FromDate_OUT=MAX('+QUOTENAME(@DateColumn)+')
    FROM '+QUOTENAME(@TargetDB)+'.'+QUOTENAME(@TargetSchema)+'.'+QUOTENAME(@TargetTable)

    EXEC sp_executesql
        @SQL,
        N'@FromDate_OUT DATETIME OUTPUT',
        @FromDate OUTPUT

    ---------------------------------------------------
    -- Calculate To Date
    ---------------------------------------------------

    IF @RetentionType='DAYS'
        SET @ToDate=DATEADD(DAY,-@RetentionValue,@ActivityDate)
    ELSE
        SET @ToDate=DATEADD(MONTH,-@RetentionValue,@ActivityDate)

    ---------------------------------------------------
    -- Current Archival Table Count
    ---------------------------------------------------

    SET @SQL='
    SELECT @Cnt=COUNT(*)
    FROM '+QUOTENAME(@TargetDB)+'.'+QUOTENAME(@TargetSchema)+'.'+QUOTENAME(@TargetTable)

    EXEC sp_executesql
        @SQL,
        N'@Cnt BIGINT OUTPUT',
        @BeforeCount OUTPUT

    ---------------------------------------------------
    -- Rows To Archive
    ---------------------------------------------------

    SET @SQL='
    SELECT @Rows=COUNT(*)
    FROM '+QUOTENAME(@SourceDB)+'.'+QUOTENAME(@SourceSchema)+'.'+QUOTENAME(@SourceTable)+' WITH(NOLOCK)
    WHERE '+QUOTENAME(@DateColumn)+' > @FromDate
    AND '+QUOTENAME(@DateColumn)+' <= @ToDate'

    EXEC sp_executesql
        @SQL,
        N'@FromDate DATETIME,@ToDate DATETIME,@Rows BIGINT OUTPUT',
        @FromDate,
        @ToDate,
        @ArchivedRows OUTPUT

    SET @ArchivedRows=ISNULL(@ArchivedRows,0)

    SET @AfterCount=@BeforeCount+@ArchivedRows

    ---------------------------------------------------
    -- Insert Report
    ---------------------------------------------------

    INSERT INTO dbo.Monthly_Archival_Table_Details
    (
        Archival_Month,
        Activity_Date,
        Source_Table_Name,
        Target_Table_Name,
        From_Date,
        To_Date,
        Before_Archival_Target_Row_Count,
        After_Archival_Target_Row_Count,
        Archived_Rows
    )
    VALUES
    (
        DATEFROMPARTS(YEAR(@ActivityDate),MONTH(@ActivityDate),1),
        @ActivityDate,

        @SourceDB+'.'+@SourceSchema+'.'+@SourceTable,
        @TargetDB+'.'+@TargetSchema+'.'+@TargetTable,

        @FromDate,
        @ToDate,

        @BeforeCount,
        @AfterCount,
        @ArchivedRows
    )

    FETCH NEXT FROM Cur INTO
    @SourceDB,@SourceSchema,@SourceTable,
    @TargetDB,@TargetSchema,@TargetTable,
    @DateColumn,@RetentionType,@RetentionValue
END

CLOSE Cur
DEALLOCATE Cur

END
GO


CREATE OR ALTER PROCEDURE DBADB.dbo.usp_Send_Monthly_Archival_Report
AS
BEGIN
    SET NOCOUNT ON;

    DECLARE
        @Subject NVARCHAR(500),
        @Body NVARCHAR(MAX),
        @TableHTML NVARCHAR(MAX),
        @ServerName SYSNAME,
        @LatestActivityDate DATE;

    SET @ServerName = @@SERVERNAME;

    SELECT @LatestActivityDate = MAX(Activity_Date)
    FROM DBADB.dbo.Monthly_Archival_Table_Details;

    SET @Subject =
        'Archival Row Count Report From PROD Instance (' +
        @ServerName + ') - ' +
        CONVERT(VARCHAR(10),@LatestActivityDate,120);

    SET @TableHTML =
    N'<table border="1" cellpadding="5" cellspacing="0" style="border-collapse:collapse;font-family:Calibri;font-size:11pt;">
        <tr bgcolor="#D9EAD3">
            <th>Archival Month</th>
            <th>Activity Date</th>
            <th>Source Table</th>
            <th>Target Table</th>
            <th>From Date</th>
            <th>To Date</th>
            <th>Before Archival Count</th>
            <th>Archived Rows</th>
            <th>After Archival Count</th>
        </tr>';

    SELECT
        @TableHTML = @TableHTML +
        '<tr>' +
        '<td>' + ISNULL(CAST(Archival_Month AS VARCHAR(50)),'') + '</td>' +
        '<td>' + CONVERT(VARCHAR(10),Activity_Date,120) + '</td>' +
        '<td>' + Source_Table_Name + '</td>' +
        '<td>' + Target_Table_Name + '</td>' +
        '<td>' + CONVERT(VARCHAR(10),From_Date,120) + '</td>' +
        '<td>' + CONVERT(VARCHAR(10),To_Date,120) + '</td>' +
        '<td align="right">' + FORMAT(Before_Archival_Target_Row_Count,'N0') + '</td>' +
        '<td align="right">' + FORMAT(Archived_Rows,'N0') + '</td>' +
        '<td align="right">' + FORMAT(After_Archival_Target_Row_Count,'N0') + '</td>' +
        '</tr>'
    FROM DBADB.dbo.Monthly_Archival_Table_Details
    WHERE Activity_Date = @LatestActivityDate
    ORDER BY Source_Table_Name;

    SET @TableHTML = @TableHTML + '</table>';

    SET @Body =
    '
    <html>
    <body style="font-family:Calibri;font-size:11pt;">
        <p>Hello Team,</p>

        <p>Please find below the Monthly Archival Row Count Report.</p>

        <p>
        <b>Server Name :</b> ' + @ServerName + '<br>
        <b>Activity Date :</b> ' + CONVERT(VARCHAR(10),@LatestActivityDate,120) + '<br>
        <b>Report Generated Time :</b> ' + CONVERT(VARCHAR(19),GETDATE(),120) + '
        </p>

        ' + @TableHTML + '

        <br><br>
        Thanks,<br>
        SQL DBA Team
    </body>
    </html>';

    EXEC msdb.dbo.sp_send_dbmail
        @profile_name = 'DBA_Profile',
        @recipients = 'dba_team@company.com',
        @subject = @Subject,
        @body = @Body,
        @body_format = 'HTML';
END
GO
